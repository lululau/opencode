## 概述

OpenCode 采用多层性能优化策略，确保快速启动和流畅运行：

- **启动性能**：快速初始化和热加载
- **运行时性能**：高效的事件处理和并发执行
- **内存优化**：缓存管理和内存限制
- **网络优化**：请求节流和批量处理
- **渲染性能**：虚拟滚动和增量更新

### 性能目标

| 指标         | 目标值  | 说明               |
| ------------ | ------- | ------------------ |
| 启动时间     | < 2s    | 从命令到就绪的时间 |
| 首次响应延迟 | < 500ms | 用户输入到 AI 响应 |
| 内存使用     | < 500MB | 常规工作负载       |
| 文件搜索响应 | < 300ms | 小型到中型项目     |
| 消息渲染延迟 | < 100ms | 单条消息的渲染时间 |

### Worker 线程池

使用 Worker 并行处理 CPU 密集型任务：

```typescript
// packages/ui/src/context/worker-pool.tsx
class DiffWorkerPool {
  private workers: Worker[] = []
  private queue: (() => Promise<void>)[] = []

  constructor(concurrency: number = 4) {
    for (let i = 0; i < concurrency; i++) {
      const worker = new Worker(new URL("./worker.ts", import.meta.url), { type: "module" })
      this.workers.push(worker)
    }
  }

  async process(data: DiffInput): Promise<DiffOutput> {
    return new Promise((resolve) => {
      this.queue.push(async () => {
        const worker = this.workers.shift()!
        try {
          const result = await worker.postMessage(data)
          resolve(result)
        } finally {
          this.workers.push(worker)
        }
      })
      this.flushQueue()
    })
  }
}
```

```typescript
const QUOTA_ERROR = "QuotaExceededError"

async function writeWithQuota(key: string, value: any): Promise<void> {
  try {
    await Storage.write(key, value)
  } catch (error) {
    if (isQuotaError(error)) {
      // 驱逐策略：移除最大的条目
      await evictLargestEntries()
      // 重试
      return writeWithQuota(key, value)
    }
    throw error
  }
}

function isQuotaError(error: unknown): boolean {
  if (error instanceof DOMException) {
    return (
      error.name === "QuotaExceededError" ||
      error.code === 22 || // NS_ERROR_DOM_QUOTA_REACHED
      error.name === "NS_ERROR_DOM_QUOTA_REACHED"
    )
  }
  return false
}
```

### LRU+TTL 缓存

```typescript
// 实现 LRU+TTL 混合驱逐策略
interface CacheOptions {
  maxEntries: number
  maxBytes: number
  ttl: number // 毫秒
}

class LRUTTLCache<K, V> {
  private cache = new Map<K, { value: V; expires: number; size: number }>()
  private accessOrder: K[] = []
  private totalBytes = 0

  get(key: K): V | undefined {
    const entry = this.cache.get(key)
    if (!entry) return undefined

    // 检查 TTL
    if (Date.now() > entry.expires) {
      this.delete(key)
      return undefined
    }

    // 更新访问顺序
    this.touch(key)
    return entry.value
  }

  set(key: K, value: V): void {
    const size = this.getSize(value)
    const expires = Date.now() + this.options.ttl

    // 检查限制
    if (this.cache.size >= this.options.maxEntries || this.totalBytes + size > this.options.maxBytes) {
      this.evict()
    }

    // 删除已存在的（更新大小）
    this.delete(key)

    this.cache.set(key, { value, expires, size })
    this.touch(key)
    this.totalBytes += size
  }

  private evict(): void {
    // 策略：驱逐最久未访问的
    const lruKey = this.accessOrder.shift()!
    const entry = this.cache.get(lruKey)!
    this.cache.delete(lruKey)
    this.totalBytes -= entry.size
  }

  private touch(key: K): void {
    // 移到访问顺序末尾
    this.accessOrder = this.accessOrder.filter((k) => k !== key)
    this.accessOrder.push(key)
  }
}
```

### 视图修剪

```typescript
// packages/app/src/context/file.tsx
const MAX_VIEW_FILES = 500
const MAX_FILE_VIEW_SESSIONS = 20

class FileViewManager {
  private views = new Map<string, FileView>()

  setView(id: string, view: FileView) {
    // 修剪超过限制的视图
    if (this.views.size >= MAX_FILE_VIEW_SESSIONS) {
      const oldest = this.getOldestView()
      this.views.delete(oldest.id)
    }

    this.views.set(id, view)
  }

  pruneFiles(): void {
    const allFiles = this.getAllFiles()
    if (allFiles.length <= MAX_VIEW_FILES) return

    // 保留当前文件，修剪其他
    const toKeep = new Set([this.currentFileId])
    const toRemove = allFiles.filter((f) => !toKeep.has(f.id))

    toRemove.forEach((f) => this.views.delete(f.id))
  }
}
```

### 请求取消

使用 `AbortController` 取消过时请求：

```typescript
let currentRequest: AbortController | null = null

async function fetchWithCancel(url: string): Promise<Response> {
  // 取消上一个请求
  currentRequest?.abort()
  currentRequest = new AbortController()

  try {
    const response = await fetch(url, {
      signal: currentRequest.signal,
    })
    return response
  } catch (error) {
    if (error.name === "AbortError") {
      // 请求被取消，静默忽略
      return null as any
    }
    throw error
  }
}
```

### 连接池

```typescript
// SSE 连接池
class SSEConnectionPool {
  private connections = new Map<string, EventSource>()
  private maxConnections = 5

  async getConnection(sessionID: string): Promise<EventSource> {
    // 检查是否已有连接
    if (this.connections.has(sessionID)) {
      return this.connections.get(sessionID)!
    }

    // 创建新连接
    const connection = new EventSource(`/session/${sessionID}/events`)

    // 管理连接数
    if (this.connections.size >= this.maxConnections) {
      const oldest = this.getOldestConnection()
      oldest?.close()
      this.connections.delete(oldest.sessionID)
    }

    this.connections.set(sessionID, connection)
    return connection
  }
}
```

### 请求批处理

```typescript
// 合并多个请求为单个批量请求
class BatchRequestManager {
  private pending: Map<string, any> = new Map()
  private flushTimer: number | null = null

  add(key: string, value: any): void {
    this.pending.set(key, value)

    // 延迟 flush
    if (this.flushTimer) return

    this.flushTimer = setTimeout(() => {
      this.flush()
    }, 50)
  }

  private async flush(): Promise<void> {
    const batch = Object.fromEntries(this.pending)
    this.pending.clear()
    this.flushTimer = null

    // 批量执行
    await executeBatchRequest(batch)
  }
}
```

### IntersectionObserver 滚动追踪

```typescript
// 替换 querySelectorAll 扫描
class ScrollSpy {
  private observer: IntersectionObserver
  private visibleElements = new Set<Element>()

  constructor() {
    this.observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          const element = entry.target as HTMLElement

          if (entry.isIntersecting) {
            this.visibleElements.add(element)
            this.onElementVisible(element.dataset.messageId!)
          } else {
            this.visibleElements.delete(element)
            this.onElementHidden(element.dataset.messageId!)
          }
        }
      },
      { root: scrollContainer, threshold: [0, 1] },
    )
  }

  observe(elements: HTMLElement[]): void {
    for (const element of elements) {
      this.observer.observe(element)
    }
  }

  // 二分查找预计算偏移量
  getVisibleIndex(): number {
    const array = Array.from(this.visibleElements)
    // 二分查找索引
    return this.binarySearch(array, scrollTop)
  }

  private binarySearch(arr: Element[], target: number): number {
    let left = 0
    let right = arr.length

    while (left < right) {
      const mid = Math.floor((left + right) / 2)
      const offset = arr[mid].offsetY

      if (offset < target) {
        left = mid + 1
      } else {
        right = mid
      }
    }

    return left
  }
}
```
